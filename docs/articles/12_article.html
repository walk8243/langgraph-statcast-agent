-- [素材 1: 素材①: 基本打撃指標素材]
-- 指示: 鈴木誠也の2024年、2025年、2026年シーズンにおける PA, AB, H, HR, RBI, R, BB%, K%, AVG, OBP, SLG, OPS を集計
SELECT
    g.season AS season,
    p.full_name AS player_name,
    sum(bb.plate_appearances) AS pa,
    sum(bb.at_bats) AS ab,
    sum(bb.hits) AS h,
    sum(bb.home_runs) AS hr,
    sum(bb.rbi) AS rbi,
    sum(bb.runs) AS r,
    round(sum(bb.base_on_balls) / nullIf(sum(bb.plate_appearances), 0), 3) AS bb_pct,
    round(sum(bb.strike_outs) / nullIf(sum(bb.plate_appearances), 0), 3) AS k_pct,
    round(sum(bb.hits) / nullIf(sum(bb.at_bats), 0), 3) AS avg,
    round((sum(bb.hits) + sum(bb.base_on_balls) + sum(bb.hit_by_pitch)) / nullIf(sum(bb.at_bats) + sum(bb.base_on_balls) + sum(bb.hit_by_pitch) + sum(bb.sac_flies), 0), 3) AS obp,
    round(sum(bb.total_bases) / nullIf(sum(bb.at_bats), 0), 3) AS slg,
    round(obp + slg, 3) AS ops
FROM statcast.boxscore_batting bb FINAL
JOIN statcast.players p FINAL ON bb.player_id = p.player_id
JOIN statcast.games g FINAL ON toUInt64(bb.game_pk) = g.game_pk
WHERE g.season IN (2024, 2025, 2026)
  AND g.game_type = 'R'
  AND p.full_name LIKE '%Seiya Suzuki%'
GROUP BY g.season, p.player_id, p.full_name
ORDER BY g.season ASC

-- [素材 2: 素材②: 打球品質・高度指標素材]
-- 指示: 鈴木誠也の2024年、2025年、2026年シーズンにおける 平均打球初速, HardHit%, Barrel%, SweetSpot%, xBA, xwOBA を集計
SELECT
    g.season AS season,
    count(s.launch_speed) AS batted_balls,
    round(avg(s.launch_speed), 1) AS avg_launch_speed_mph,
    round(countIf(s.launch_speed >= 95.0) * 100.0 / count(s.launch_speed), 1) AS hard_hit_pct,
    round(countIf(s.launch_speed_angle = 6) * 100.0 / count(s.launch_speed), 1) AS barrel_pct,
    round(countIf(s.launch_angle BETWEEN 8 AND 32) * 100.0 / count(s.launch_speed), 1) AS sweet_spot_pct,
    round(avg(s.estimated_ba_using_speedangle), 3) AS xba,
    round(avg(s.estimated_woba_using_speedangle), 3) AS xwoba
FROM statcast.statcast_raw AS s
JOIN statcast.players p FINAL ON toUInt64(s.batter) = p.player_id
JOIN statcast.games g FINAL ON toUInt64(s.game_pk) = g.game_pk
WHERE g.season IN (2024, 2025, 2026)
  AND g.game_type = 'R'
  AND s.type = 'X'
  AND s.launch_speed IS NOT NULL
  AND p.full_name LIKE '%Seiya Suzuki%'
GROUP BY g.season
ORDER BY g.season ASC

-- [素材 3: 素材③: 配球・球種別適応素材]
-- 指示: 鈴木誠也の2024年、2025年、2026年シーズンにおける 球種別（Fastball, Offspeed, Breaking）の打率、長打率、Whiff%（空振り率）を集計
WITH pitch_categorized AS (
    SELECT
        s.game_pk,
        s.batter,
        s.pitch_type,
        s.pitch_name,
        s.description,
        s.events,
        CASE
            WHEN s.pitch_type IN ('FF', 'FT', 'SI', 'FC', 'FS', 'FA') THEN 'Fastball'
            WHEN s.pitch_type IN ('CH', 'FO', 'SC', 'EP') THEN 'Offspeed'
            ELSE 'Breaking'
        END AS pitch_category
    FROM statcast.statcast_raw AS s
    JOIN statcast.games g FINAL ON toUInt64(s.game_pk) = g.game_pk
    WHERE g.season IN (2024, 2025, 2026)
      AND g.game_type = 'R'
      AND toUInt64(s.batter) IN (
          SELECT player_id FROM statcast.players FINAL WHERE full_name LIKE '%Seiya Suzuki%'
      )
),
player_info AS (
    SELECT player_id FROM statcast.players FINAL WHERE full_name LIKE '%Seiya Suzuki%'
),
batting_events AS (
    SELECT
        g.season,
        pc.pitch_category,
        countIf(pc.events IN ('single', 'double', 'triple', 'home_run')) AS hits,
        countIf(pc.events IN ('single', 'double', 'triple', 'home_run', 'field_out', 'strikeout', 'force_out', 'grounded_into_double_play', 'sac_fly', 'sac_bunt') OR pc.events != '') AS at_bats_approx,
        countIf(pc.events = 'single') * 1 + countIf(pc.events = 'double') * 2 + countIf(pc.events = 'triple') * 3 + countIf(pc.events = 'home_run') * 4 AS total_bases_approx,
        countIf(pc.description IN ('swinging_strike', 'swinging_strike_blocked')) AS swinging_strikes,
        countIf(pc.description IN ('swinging_strike', 'swinging_strike_blocked', 'foul', 'hit_into_play', 'foul_bunt', 'missed_bunt')) AS total_swings
    FROM pitch_categorized AS pc
    JOIN statcast.games g FINAL ON toUInt64(pc.game_pk) = g.game_pk
    GROUP BY g.season, pc.pitch_category
)
SELECT
    season,
    pitch_category,
    round(hits / nullIf(at_bats_approx, 0), 3) AS avg,
    round(total_bases_approx / nullIf(at_bats_approx, 0), 3) AS slg,
    round(swinging_strikes * 100.0 / nullIf(total_swings, 0), 1) AS whiff_pct
FROM batting_events
WHERE at_bats_approx > 0
ORDER BY season DESC, pitch_category ASC

進化し続ける鈴木誠也：2024-2026年データが証明する「真の打者への階段」

概要・エグゼクティブサマリー

メジャーリーグの舞台で確固たる地位を築きつつある鈴木誠也。彼が2024年から2026年にかけて歩んできた3シーズンは、単なる「数字の推移」に留まらず、打者としてのモデルチェンジと進化の連続でした。伝統的な打撃指標から最先端のStatcast（スタットキャスト）データ、さらには配球別の適応プロセスに至るまで、多角的なアプローチからその実態を紐解くと、彼の打撃が「いかにしてメジャーのトップクラスへ近づいているか」が鮮明に浮かび上がります。

本記事の主要なハイライト：
- 長打力の開花と安定感: 2025年には自己最多の32本塁打、103打点をマークし、パワーヒッターとしての存在感を完全に確立。
- 打球プロファイルの変遷: 2024年の圧倒的な打球品質から、2025年には球界トップクラスのバレル率（16.6%）を記録するなど、打球角度とパワーの融合を実現。
- アプローチの深化: 選球眼の向上（BB%の向上）が見られる一方で、球種ごとの適応やウィークポイントの克服に向けた絶え間ない試行錯誤がデータに表れている。

2024シーズン、鈴木は585打席に立ち、打率.283、21本塁打、OPS .848という非常にバランスの取れた素晴らしい成績を残した。四球率（BB%）10.8%に対して三振率（K%）27.4%という数字は、やや三振が目立つものの、高い出塁率（.366）がそれを補う形であった。

続く2025シーズン、打席数は651へと大きく増加し、フルシーズンを通して中軸としての責任を全うした。特筆すべきは本塁打数の激増である。自己最多となる32本塁打、そして103打点をマークし、打撃の軸足が「アベレージヒッター」から「量産型スラッガー」へとシフトしたことを証明した。打率は.245へと低下したものの、長打率.478、OPS .804と高い生産性を維持している。

そして2026シーズンは、打率が.270へと再び上昇し、四球率も12.8%へ向上。三振率も24.4%へと低下し、アプローチの確実性がさらに増した。本塁打こそ26本と2025年から微減したものの、得点（R）が96へと大きく跳ね上がり、打線の中核としてチームの勝利に深く貢献している姿が数字から読み取れる。

第2章：Statcastが暴く「打球品質」の劇的進化

続いて、打者の真のポテンシャルを測るStatcastの「打球品質・高度指標」を検証する。これにより、当たり損ないの傾向や、どれだけ効率的に強い打球を放っているかが明らかになる。

このデータテーブルは、鈴木の打撃における「質」のダイナミックな変化を雄弁に語っている。

2024年は、平均打球初速が91.7 mph（約147.6 km/h）、HardHit%（95 mph以上の打球の割合）が49.3%と、リーグ屈指のハードヒッターとしての資質を遺憾なく発揮していた。expected指標であるxBA（.367）やxwOBA（.442）が非常に高く、不運な当たりが多かったことを示唆している。

さらに驚異的なのが2025シーズンだ。打球数が415に増えた中で、バレル率（Barrel%）が16.6%という驚異的な数値に跳ね上がった。バレルとは「最適な打球角度と初速で放たれ、長打になる確率が非常に高い打球」を指すが、16%台という数字はメジャーリーガーの中でもトッププロスペクト、あるいはエリートスラッガーの領域である。スウィートスポット率（SweetSpot%）も42.2%へと向上しており、「いかに芯で捉え、理想的な角度でボールを上げるか」というメカニズムの深化が成功したシーズンだったと言える。

2026シーズンに入ると、平均打球初速は88.6 mph、HardHit%は38.9%、バレル率も10.7%へとやや下降線を描いている。しかし、これは単なるパワーの衰えではなく、第1章で見たように三振率の低下（24.4%）や四球率の向上（12.8%）と連動しており、「力任せに引っ張る」スタイルから、よりコンタクト重視、あるいは状況に応じた柔軟なアプローチへと微調整を行った結果である可能性が高い。

まずFastball（速球系）に対するアプローチを見ると、Whiff%（空振り率）が2024年の18.4%から、2025年（15.6%）、2026年（15.4%）へと一貫して低下している。これはメジャーの剛速球に対するタイミグの合わせ方がより確実なものになり、差し込まれるケースが減っている証拠だ。

一方で、Breaking（変化球系）に対しては苦戦の痕跡が見える。2024年には打率.235、長打率.382だったものが、2025年には打率.178、長打率.322まで落ち込んだ。2026年には打率.205、長打率.368と持ち直しているものの、空振り率（28.8%〜31.2%前後）も含め、メジャー特有の鋭く曲がり落ちるスライダーやカーブへの対応は継続的な課題であり、相手バッテリーもここに活路を見出そうとしていることがうかがえる。

最も特筆すべきはOffspeed（オフスピード系：チェンジアップやフォークなど）の進化だ。2024年は長打率.288にとどまっていたが、2025年には長打率.576、2026年も長打率.556と、ゾーンに来たオフスピード系を完璧に捉えて長打にする能力が劇的に向上している。変化球でタイミングを外すアプローチに対し、鈴木は「落ちる球や緩い球を我慢して呼び込み、長打にする」という技術をこの3年間で完全に自分のものにしたと言える。

まとめと今後の展望

2024年から2026年までの3シーズンにわたるデータを総括すると、鈴木誠也という打者は単なる「パワーヒッターへの変貌」を遂げただけでなく、「状況や配球に応じた高度なアジャスト能力を持つ成熟した打者」へと階段を駆け上がっていることが証明された。

  • 2024年: 高い打球初速と優れたxBAが示す通り、メジャーの環境への適応とベースとなる高い打撃素質を証明。
  • 2025年: バレル率16.6%に代表される爆発的な打球品質の向上により、32本塁打・103打点というキャリアハイを達成。
  • 2026年: パワー偏重から脱却し、三振率の低下、選球眼の向上、そしてオフスピード系に対する圧倒的な長打力（SLG .556）を手に入れ、より総合力の高い打者へ進化。

変化球（Breaking）に対する空振り率のコントロールやさらなるアベレージの向上といった伸び代を残しつつも、データが示す彼の進化の軌跡は実にエキサイティングである。相手バッテリーがどのような配球で攻めてこようとも、それをデータで分析し、自らのスイングを微調整して適応していく――。鈴木誠也がメジャーリーグの真の「強打者（The Elite Hitter）」として君臨する未来は、すでに数字の裏付けをもって現実のものとなっている。
