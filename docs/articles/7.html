SELECT
    p.full_name AS player_name,
    count() AS total_batted_balls,
    countIf(s.events = 'home_run') AS home_runs,
    countIf(s.events IN ('single', 'double', 'triple', 'home_run')) AS hits,
    round(avg(s.launch_speed), 1) AS avg_exit_velocity,
    round(max(s.launch_speed), 1) AS max_exit_velocity,
    round(avg(s.launch_angle), 1) AS avg_launch_angle,
    round(avg(if(s.launch_speed >= 95, 1, 0)) * 100, 1) AS hard_hit_percentage
FROM statcast.statcast_raw AS s
JOIN statcast.players AS p ON toUInt64(s.batter) = p.player_id
JOIN statcast.games AS g ON toUInt64(s.game_pk) = g.game_pk
WHERE g.season = 2026 
  AND g.game_type = 'R'
  AND p.full_name IN ('Seiya Suzuki', 'Munetaka Murakami', 'Kazuma Okamoto')
  AND s.launch_speed IS NOT NULL
GROUP BY player_name
ORDER BY home_runs DESC, avg_exit_velocity DESC
