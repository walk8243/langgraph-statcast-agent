-- [素材 1: 素材①: 2026年基本打撃スタッツ]
-- 指示: 2026年シーズンの鈴木誠也、村上宗隆、岡本和真のPA（打席数）, AB（打数）, H（安打）, HR（本塁打）, RBI（打点）, AVG（打率）, OBP（出塁率）, SLG（長打率）, OPS, wOBAを集計
SELECT
    p.full_name AS player_name,
    sum(bb.plate_appearances) AS pa,
    sum(bb.at_bats) AS ab,
    sum(bb.hits) AS h,
    sum(bb.home_runs) AS hr,
    sum(bb.rbi) AS rbi,
    round(sum(bb.hits) / nullIf(sum(bb.at_bats), 0), 3) AS avg,
    round((sum(bb.hits) + sum(bb.base_on_balls) + sum(bb.hit_by_pitch)) / nullIf(sum(bb.at_bats) + sum(bb.base_on_balls) + sum(bb.hit_by_pitch) + sum(bb.sac_flies), 0), 3) AS obp,
    round(sum(bb.total_bases) / nullIf(sum(bb.at_bats), 0), 3) AS slg,
    round(obp + slg, 3) AS ops,
    round(avg(s.estimated_woba_using_speedangle), 3) AS xwoba
FROM statcast.boxscore_batting AS bb
JOIN statcast.players p FINAL ON bb.player_id = p.player_id
JOIN statcast.games g FINAL ON toUInt64(bb.game_pk) = g.game_pk
LEFT JOIN statcast.statcast_raw s ON toUInt64(bb.game_pk) = toUInt64(s.game_pk) AND bb.player_id = toUInt64(s.batter)
WHERE g.season = 2026
  AND g.game_type = 'R'
  AND (p.full_name LIKE '%Seiya Suzuki%' OR p.full_name LIKE '%Munetaka Murakami%' OR p.full_name LIKE '%Kazuma Okamoto%')
GROUP BY p.player_id, p.full_name
ORDER BY ops DESC

-- [素材 2: 素材②: Statcast打球品質データ]
-- 指示: 2026年シーズンの鈴木誠也、村上宗隆、岡本和真の平均打球初速（Avg Exit Velocity）、最大打球初速（Max Exit Velocity）、バレル率（Barrel%）、スイートスポット率（Sweet Spot%）、ハードヒット率（HardHit%）を集計
SELECT
    p.full_name AS player_name,
    count() AS total_batted_balls,
    round(avg(s.launch_speed), 1) AS avg_launch_speed_mph,
    round(max(s.launch_speed), 1) AS max_launch_speed_mph,
    round(countIf(s.launch_speed >= 95.0) * 100.0 / count(), 1) AS hard_hit_pct,
    round(countIf(s.launch_speed_angle = 6) * 100.0 / count(), 1) AS barrel_pct,
    round(countIf(s.launch_angle BETWEEN 8 AND 32) * 100.0 / count(), 1) AS sweet_spot_pct
FROM statcast.statcast_raw AS s
JOIN statcast.players p FINAL ON toUInt64(s.batter) = p.player_id
JOIN statcast.games g FINAL ON toUInt64(s.game_pk) = g.game_pk
WHERE g.season = 2026
  AND g.game_type = 'R'
  AND s.type = 'X'
  AND s.launch_speed IS NOT NULL
  AND (p.full_name LIKE '%Seiya Suzuki%' OR p.full_name LIKE '%Munetaka Murakami%' OR p.full_name LIKE '%Kazuma Okamoto%')
GROUP BY p.full_name
ORDER BY avg_launch_speed_mph DESC

-- [素材 3: 素材③: 球種別パフォーマンスと空振り率]
-- 指示: 2026年シーズンの鈴木誠也、村上宗隆、岡本和真の対速球（4-Seam）、対変化球（Breaking）、対オフスピード（Offspeed）におけるwOBA、打率、およびWhiff%（空振り率）を集計
SELECT
    p.full_name AS player_name,
    CASE
        WHEN s.pitch_name = '4-Seam Fastball' THEN 'Fastball'
        WHEN s.pitch_type IN ('CU', 'SL', 'SV', 'KC', 'CS') THEN 'Breaking'
        WHEN s.pitch_type IN ('CH', 'FS', 'SC', 'FO') THEN 'Offspeed'
        ELSE 'Other'
    end AS pitch_category,
    round(sum(s.woba_value) / nullIf(sum(s.woba_denom), 0), 3) AS woba,
    round(countIf(s.events IN ('single', 'double', 'triple', 'home_run')) / nullIf(countIf(s.events IN ('single', 'double', 'triple', 'home_run', 'field_out', 'force_out', 'grounded_into_double_play')), 0), 3) AS avg,
    round(countIf(s.description = 'swinging_strike') * 100.0 / nullIf(countIf(s.description LIKE '%swing%'), 0), 2) AS whiff_pct
FROM statcast.statcast_raw AS s
JOIN statcast.players p FINAL ON toUInt64(s.batter) = p.player_id
JOIN statcast.games g FINAL ON toUInt64(s.game_pk) = g.game_pk
WHERE g.season = 2026
  AND g.game_type = 'R'
  AND (p.full_name LIKE '%Seiya Suzuki%' OR p.full_name LIKE '%Munetaka Murakami%' OR p.full_name LIKE '%Kazuma Okamoto%')
  AND (
      s.pitch_name = '4-Seam Fastball'
      OR s.pitch_type IN ('CU', 'SL', 'SV', 'KC', 'CS', 'CH', 'FS', 'SC', 'FO')
  )
GROUP BY p.full_name, pitch_category
ORDER BY p.full_name, pitch_category

概要・エグゼクティブサマリー

メジャーリーグベースボール（MLB）の舞台で、日本を代表する強打者たちがどのような打撃を見せているのか。2026年シーズンにおける 鈴木誠也、村上宗隆、岡本和真 の3選手に焦点を当て、伝統的な打撃指標から最新のStatcastデータまでを多角的に分析・比較する。

本記事の主要な発見・ハイライトは以下の通り：
- 鈴木誠也 は、安定した出塁率と高いOPS（0.838）を誇り、特に速球に対する圧倒的な対応力（wOBA 0.444）でチームを牽引する。

- 岡本和真 は、優れたスイートスポット率（36.1%）に裏打ちされたコンタクトの安定性と、変化球への鋭いアプローチが光る。
